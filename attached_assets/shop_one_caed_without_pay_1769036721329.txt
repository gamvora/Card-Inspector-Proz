# Helper function equivalent to PHP's find_between
def find_between(content, start, end):
    try:
        start_pos = content.index(start) + len(start)
        end_pos = content.index(end, start_pos)
        return content[start_pos:end_pos]
    except ValueError:
        return ''

# Function to make requests with proxy and retry logic
def make_request_with_proxy(url, method='GET', headers=None, json_data=None, data=None, cookies=None, timeout=15):
    global retry_count
    global max_retries
    
    proxy = get_proxy()
    
    for attempt in range(max_retries):
        try:
            if method.upper() == 'GET':
                response = requests.get(url, headers=headers, proxies=proxy, 
                                      verify=False, timeout=timeout, cookies=cookies)
            elif method.upper() == 'POST':
                response = requests.post(url, headers=headers, json=json_data, 
                                       data=data, proxies=proxy, verify=False, 
                                       timeout=timeout, cookies=cookies)
            else:
                raise ValueError(f"Unsupported HTTP method: {method}")
            
            response.raise_for_status()
            return response
            
        except (requests.exceptions.RequestException, requests.exceptions.Timeout) as e:
            retry_count += 1
            print(f"[Retry {retry_count}/{max_retries}] Proxy error for {url}: {e}")
            
            # Rotate to a new proxy by waiting a bit (simulating rotation)
            time.sleep(1)
            
            if attempt == max_retries - 1:
                raise Exception(f"Failed after {max_retries} attempts: {e}")

# Function to read CC from file
def read_cc_from_file(filename):
    try:
        with open(filename, 'r') as file:
            lines = file.readlines()
            cc_list = []
            for line in lines:
                line = line.strip()
                if line and '|' in line:
                    cc_parts = line.split('|')
                    if len(cc_parts) >= 4:
                        cc_list.append({
                            'cc': cc_parts[0],
                            'month': cc_parts[1],
                            'year': cc_parts[2],
                            'cvv': cc_parts[3]
                        })
            return cc_list
    except FileNotFoundError:
        print(f"File {filename} not found.")
        return []
    except Exception as e:
        print(f"Error reading file: {e}")
        return []
